Option Explicit
' Paste into the CODE WINDOW of a new blank UserForm named frmUSTANodeCleaner.
' Controls are created at runtime using built-in MSForms (no .frm import).

Private WithEvents btnClean As MSForms.CommandButton
Private WithEvents btnCancel As MSForms.CommandButton
Private WithEvents btnDefaults As MSForms.CommandButton
Private WithEvents optQuick As MSForms.OptionButton
Private WithEvents optMedium As MSForms.OptionButton
Private WithEvents optSerious As MSForms.OptionButton
Private WithEvents optExtreme As MSForms.OptionButton
Private WithEvents txtAttempts As MSForms.TextBox
Private txtTolerance As MSForms.TextBox
Private updating As Boolean

Private Sub UserForm_Initialize()
    Dim bg As Long, panel As Long, accent As Long, muted As Long
    Dim clientW As Single, clientH As Single
    Dim margin As Single, contentW As Single, gap As Single, colW As Single
    Dim leftX As Single, rightX As Single, inputX As Single
    Dim fieldW As Single, btnGap As Single, sideBtnW As Single, mainBtnW As Single
    Dim buttonY As Single

    bg = RGB(32, 36, 44)
    panel = RGB(47, 53, 63)
    accent = RGB(255, 174, 66)
    muted = RGB(188, 197, 210)

    Me.Caption = "USTA Smart Node Cleaner v0.7"
    Me.BackColor = bg
    Me.StartUpPosition = 1

    ' Set the desired CLIENT area, not the outer window size.
    ' Windows borders/title bar vary with theme and display scaling.
    Me.Width = 420
    Me.Height = 405
    clientW = 410
    clientH = 380
    Me.Width = clientW + (Me.Width - Me.InsideWidth)
    Me.Height = clientH + (Me.Height - Me.InsideHeight)

    ' Every horizontal coordinate derives from the actual client width.
    margin = 22
    gap = 12
    contentW = Me.InsideWidth - 2 * margin
    colW = (contentW - gap) / 2
    leftX = margin
    rightX = leftX + colW + gap
    fieldW = 78
    inputX = leftX + contentW - fieldW - 29  ' reserve 29 pt for "mm"

    AddLabel "brand", "USTA  /  SMART NODE CLEANER", leftX, 16, contentW, 28, accent, 16, True
    AddLabel "subline", "VECTOR TOOLS   |   CorelDRAW 2018   |   v0.7", leftX, 47, contentW, 16, muted, 9, False

    AddLabel "precision", "01   PRECISION", leftX, 77, contentW, 18, accent, 10, True
    AddLabel "tolcaption", "Maximum sampled deviation", leftX, 104, inputX - leftX - 8, 20, muted, 10, False
    Set txtTolerance = Me.Controls.Add("Forms.TextBox.1", "txtTolerance", True)
    StyleText txtTolerance, inputX, 101, fieldW, 24, panel
    txtTolerance.Text = "0.05"
    AddLabel "mm", "mm", inputX + fieldW + 8, 105, 22, 19, muted, 10, False

    AddLabel "workload", "02   WORKLOAD / MAXIMUM ATTEMPTS", leftX, 146, contentW, 20, accent, 10, True
    Set optQuick = AddOption("optQuick", "QUICK    512", leftX, 173, panel, muted, colW)
    Set optMedium = AddOption("optMedium", "MEDIUM    1024", rightX, 173, panel, muted, colW)
    Set optSerious = AddOption("optSerious", "SERIOUS    2048", leftX, 207, panel, muted, colW)
    Set optExtreme = AddOption("optExtreme", "EXTREME    4096", rightX, 207, panel, muted, colW)

    AddLabel "maxlabel", "Maximum attempts (128 - 4096)", leftX, 249, inputX - leftX - 8, 20, muted, 10, False
    Set txtAttempts = Me.Controls.Add("Forms.TextBox.1", "txtAttempts", True)
    StyleText txtAttempts, inputX, 246, fieldW, 24, panel

    AddLabel "note1", "Upper limit only; stops when no nodes remain to test.", leftX, 283, contentW, 16, muted, 9, False
    AddLabel "note2", "Select one curve to clean. Ctrl+Z to undo.", leftX, 301, contentW, 16, muted, 9, False

    ' Bottom row anchored to actual client height.
    buttonY = Me.InsideHeight - 42
    btnGap = 12
    sideBtnW = 108
    mainBtnW = contentW - 2 * sideBtnW - 2 * btnGap
    Set btnDefaults = AddButton("btnDefaults", "DEFAULTS", leftX, buttonY, sideBtnW, 30, panel, vbWhite)
    Set btnClean = AddButton("btnClean", "CLEAN NODES", leftX + sideBtnW + btnGap, buttonY, mainBtnW, 30, accent, RGB(20, 24, 30))
    Set btnCancel = AddButton("btnCancel", "CANCEL", leftX + contentW - sideBtnW, buttonY, sideBtnW, 30, panel, vbWhite)
    btnClean.Default = True
    btnCancel.Cancel = True

    ApplyDefaults
End Sub

Private Function AddLabel(ByVal key As String, ByVal caption As String, _
    ByVal x As Single, ByVal y As Single, ByVal w As Single, ByVal h As Single, _
    ByVal color As Long, ByVal size As Single, ByVal isBold As Boolean) As MSForms.Label
    Dim c As MSForms.Label
    Set c = Me.Controls.Add("Forms.Label.1", key, True)
    c.Caption = caption
    c.Left = x: c.Top = y: c.Width = w: c.Height = h
    c.BackStyle = fmBackStyleTransparent
    c.ForeColor = color
    c.Font.Name = "Tahoma": c.Font.Size = size: c.Font.Bold = isBold
    Set AddLabel = c
End Function

Private Sub StyleText(ByVal c As MSForms.TextBox, ByVal x As Single, _
    ByVal y As Single, ByVal w As Single, ByVal h As Single, ByVal bg As Long)
    c.Left = x: c.Top = y + 2: c.Width = w: c.Height = h - 4
    c.BackColor = bg: c.ForeColor = vbWhite
    c.Font.Name = "Tahoma": c.Font.Size = 10
    c.BorderStyle = fmBorderStyleSingle
    c.TextAlign = fmTextAlignLeft
End Sub

Private Function AddOption(ByVal key As String, ByVal caption As String, _
    ByVal x As Single, ByVal y As Single, ByVal bg As Long, _
    ByVal fg As Long, ByVal optionWidth As Single) As MSForms.OptionButton
    Dim c As MSForms.OptionButton
    Set c = Me.Controls.Add("Forms.OptionButton.1", key, True)
    c.Caption = caption
    ' Draw the full-width background separately; inset the actual radio control.
    Dim tile As MSForms.Label
    Set tile = Me.Controls.Add("Forms.Label.1", key & "Tile", True)
    tile.Caption = ""
    tile.Left = x: tile.Top = y: tile.Width = optionWidth: tile.Height = 29
    tile.BackStyle = fmBackStyleOpaque
    tile.BackColor = bg

    c.Left = x + 9: c.Top = y + 2: c.Width = optionWidth - 18: c.Height = 25
    c.BackStyle = fmBackStyleTransparent
    c.ForeColor = fg
    c.Font.Name = "Tahoma": c.Font.Size = 10: c.Font.Bold = True
    c.ZOrder 0
    Set AddOption = c
End Function

Private Function AddButton(ByVal key As String, ByVal caption As String, _
    ByVal x As Single, ByVal y As Single, ByVal w As Single, _
    ByVal h As Single, ByVal bg As Long, ByVal fg As Long) As MSForms.CommandButton
    Dim c As MSForms.CommandButton
    Set c = Me.Controls.Add("Forms.CommandButton.1", key, True)
    c.Caption = caption
    c.Left = x: c.Top = y: c.Width = w: c.Height = h
    c.BackColor = bg: c.ForeColor = fg
    c.Font.Name = "Tahoma": c.Font.Size = 10: c.Font.Bold = True
    Set AddButton = c
End Function

Private Sub ApplyDefaults()
    updating = True
    txtTolerance.Text = "0.05"
    txtAttempts.Text = "512"
    optQuick.Value = True
    updating = False
End Sub

Private Sub SetWorkload(ByVal count As Long)
    If updating Then Exit Sub
    updating = True
    txtAttempts.Text = CStr(count)
    updating = False
End Sub

Private Sub optQuick_Click()
    SetWorkload 512
End Sub
Private Sub optMedium_Click()
    SetWorkload 1024
End Sub
Private Sub optSerious_Click()
    SetWorkload 2048
End Sub
Private Sub optExtreme_Click()
    SetWorkload 4096
End Sub

Private Sub txtAttempts_Change()
    If updating Then Exit Sub
    updating = True
    optQuick.Value = False
    optMedium.Value = False
    optSerious.Value = False
    optExtreme.Value = False
    updating = False
End Sub

Private Sub btnDefaults_Click()
    ApplyDefaults
End Sub

Private Sub btnCancel_Click()
    Unload Me
End Sub

Private Sub btnClean_Click()
    Dim raw As String, count As Long, tol As Double, t As String
    raw = Trim$(txtAttempts.Text)
    If Len(raw) = 0 Or Len(raw) > 4 Or raw Like "*[!0-9]*" Then
        MsgBox "Enter a whole number from 128 to 4096.", vbExclamation, "USTA"
        Exit Sub
    End If
    count = CLng(raw)
    If count < 128 Or count > 4096 Then
        MsgBox "Maximum attempts must be 128-4096.", vbExclamation, "USTA"
        Exit Sub
    End If
    t = Trim$(txtTolerance.Text)
    If Len(t) = 0 Or Not IsNumeric(Replace(t, ",", ".")) Then
        MsgBox "Enter a valid tolerance in millimeters.", vbExclamation, "USTA"
        Exit Sub
    End If
    tol = Val(Replace(t, ",", "."))
    If tol <= 0 Or tol > 2 Then
        MsgBox "Tolerance must be greater than 0 and at most 2 mm.", vbExclamation, "USTA"
        Exit Sub
    End If
    Me.Hide
    UstaCleanRun tol, count
    Unload Me
End Sub
